                                        onChange={handleChange}
                                        style={inputStyle}
                                        isInvalid={!!touched.contactPersonRole && !!errors.contactPersonRole}
                                    >
                                        {contact_role_options.map((option) => (
                                            <option key={option.value} value={option.value}>
                                                {option.label}
                                            </option>
                                        ))}
                                    </BootstrapForm.Select>
                                </InputGroup>
                                <BootstrapForm.Control.Feedback type="invalid"><ErrorMessage name="contactPersonRole" /></BootstrapForm.Control.Feedback>
                            </BootstrapForm.Group>
                        </Col>

                        <Col md={12}>
                            <BootstrapForm.Group style={formGroupStyle}>
                                <InputGroup>
                                    <InputGroup.Text style={inputGroupTextStyle}><FontAwesomeIcon icon={faEnvelope} /></InputGroup.Text>
                                    <BootstrapForm.Control
                                        type="email"
                                        name="email"
                                        placeholder="Contact Email Address"
                                        value={values.email}
                                        onChange={handleChange}
                                        style={inputStyle}
                                        isInvalid={!!touched.email && !!errors.email}
                                    />
                                </InputGroup>
                                <BootstrapForm.Control.Feedback type="invalid"><ErrorMessage name="email" /></BootstrapForm.Control.Feedback>
                            </BootstrapForm.Group>
                        </Col>

                        <Col md={12}>
                            <BootstrapForm.Group style={formGroupStyle}>
                                <InputGroup>
                                    <InputGroup.Text style={inputGroupTextStyle}><FontAwesomeIcon icon={faPhone} /></InputGroup.Text>
                                    <BootstrapForm.Control
                                        type="tel"
                                        name="phoneNumber"
                                        placeholder="Contact Person Phone Number"
                                        value={values.phoneNumber}
                                        onChange={handleChange}
                                        style={inputStyle}
                                        isInvalid={!!touched.phoneNumber && !!errors.phoneNumber}
                                    />
                                </InputGroup>
                                <BootstrapForm.Control.Feedback type="invalid"><ErrorMessage name="phoneNumber" /></BootstrapForm.Control.Feedback>
                            </BootstrapForm.Group>
                        </Col>

                        <Col md={6}>
                            <BootstrapForm.Group style={formGroupStyle} controlId="password">
                                <InputGroup>
                                    <BootstrapForm.Control
                                        type={showPassword ? "text" : "password"}
                                        name="password"
                                        placeholder="Password"
                                        value={values.password}
                                        onChange={handleChange}
                                        style={inputStyle}
                                        isInvalid={!!touched.password && !!errors.password}
                                    />
                                    <InputGroup.Text style={passwordToggleStyle} onClick={togglePasswordVisibility}>
                                        <FontAwesomeIcon icon={showPassword ? faEyeRegular : faEyeSlash} />
                                    </InputGroup.Text>
                                </InputGroup>
                                <BootstrapForm.Control.Feedback type="invalid"><ErrorMessage name="password" /></BootstrapForm.Control.Feedback>
                            </BootstrapForm.Group>
                        </Col>

                        <Col md={6}>
                            <BootstrapForm.Group style={formGroupStyle} controlId="confirmPassword">
                                <InputGroup>
                                    <BootstrapForm.Control
                                        type={showConfirmPassword ? "text" : "password"}
                                        name="confirmPassword"
                                        placeholder="Confirm Password"
                                        value={values.confirmPassword}
                                        onChange={handleChange}
                                        style={inputStyle}
                                        isInvalid={!!touched.confirmPassword && !!errors.confirmPassword}
                                    />
                                    <InputGroup.Text style={passwordToggleStyle} onClick={toggleConfirmPasswordVisibility}>
                                        <FontAwesomeIcon icon={showConfirmPassword ? faEyeRegular : faEyeSlash} />
                                    </InputGroup.Text>
                                </InputGroup>
                                <BootstrapForm.Control.Feedback type="invalid"><ErrorMessage name="confirmPassword" /></BootstrapForm.Control.Feedback>
                            </BootstrapForm.Group>
                        </Col>
                    </Row>
                );

            case 2:
                const emailForOTP = registeredEmail || user?.email;
                const displayEmail = emailForOTP ? `${emailForOTP.substring(0, 3)}***${emailForOTP.substring(emailForOTP.indexOf('@') - 2, emailForOTP.indexOf('@'))}@gmail.com` : '';
                const titleText = isRegistrationSuccess
                    ? 'Registration Complete!'
                    : 'Verify Account';
                const descriptionText = isRegistrationSuccess
                    ? `Verification successful for your organisation! We've sent a final email to **${emailForOTP}** with instructions on accessing your account and next steps.`
                    : `Enter the 6-digit OTP we have sent to **${displayEmail}**.`;

                return (
                    <>
                        <div className="d-flex flex-column align-items-center justify-content-center mb-4">
                            <h2 className="text-center mb-2" style={{ fontFamily: "body", fontSize: '1.5rem', fontWeight: "bold", color: isRegistrationSuccess ? customStyles.primaryColor : 'inherit' }}>
                                {titleText}
                            </h2>
                            <p className="text-muted mb-4 text-center" style={{ fontFamily: "body", fontSize: '0.9rem', maxWidth: '350px' }}>
                                <span dangerouslySetInnerHTML={{ __html: descriptionText }} />
                            </p>
                        </div>

                        {!isRegistrationSuccess ? (
                            <>
                                <div className='otpGroup' style={otpGroupStyle}>
                                    <OtpInput
                                        value={otp}
                                        valueLength={OTP_LENGTH}
                                        onChange={setOtp}
                                    />
                                </div>

                                {(localError || authError) && (
                                    <div className="text-danger mt-1 mb-3 small fw-bold text-center">
                                        {localError || authError}
                                    </div>
                                )}

                                <div className="text-center mt-3">
                                    <span
                                        className="text-center text-muted small"
                                        style={{ fontFamily: 'body' }}
                                    >
                                        Did not receive a code?{' '}
                                    </span>
                                    <Link
                                        onClick={handleResendCode}
                                        style={{
                                            color: '#3CB371',
                                            textDecoration: 'none',
                                            fontWeight: '600',
                                            cursor: isResendLoading || !emailForOTP ? 'not-allowed' : 'pointer',
                                            fontFamily: 'body',
                                            opacity: isResendLoading || !emailForOTP ? 0.6 : 1,
                                        }}
                                        to="#"
                                        className="small"
                                    >
                                        {isResendLoading ? 'Sending...' : 'Resend it'}
                                    </Link>
                                </div>
                            </>
                        ) : (
                            <div className="d-flex flex-column align-items-center mt-3">
                                <FontAwesomeIcon icon={faCheckCircle} size="3x" color={customStyles.primaryColor} className="mb-3" />
                            </div>
                        )}
                    </>
                );

            default:
                return null;
        }
    };

    const primaryColorForStepper = customStyles.primaryColor;

    return (
        // Main container setup for full viewport height and centering
        <div style={{
            backgroundColor: customStyles.backgroundColor,
            minHeight: "100vh",
            display: "flex",
            justifyContent: "center",
            alignItems: "center",
            padding: '20px 0'
        }}>
            <Container className="d-flex justify-content-center align-items-center h-100">
                <Card
                    className="p-4 shadow-lg"
                    style={{
                        maxWidth: '700px',
                        width: '100%',
                        maxHeight: '90vh', // Sets a maximum height for the card
                        overflowY: 'auto', // Adds scrollbar to the card if content overflows
                        borderRadius: "8px",
                        border: "none"
                    }}
                >
                    <Card.Body className="p-0">
                        <div className="text-center mb-4">
                            <img
                                src={logo}
                                alt="Obeeoma Logo"
                                style={{ height: "40px", marginBottom: "5px" }}
                            />
                            <h4 className="fw-semibold" style={{ color: "#74787C", fontSize: "16px", fontFamily: "body" }}>
                                A HAPPY MIND
                            </h4>
                        </div>

                        {/* Step-specific title and description */}
                        {activeStep === 0 && (
                            <div className="text-center mb-4">
                                <h3 className="fw-bold mb-2" style={{ fontFamily: "body", fontSize: '1.75rem' }}>Create your Organization's account</h3>
                                <p className="text-muted" style={{ fontFamily: "body", fontSize: '0.9rem' }}>
                                    Get early access to Obeeoma and empower your team with smarter mental health support.
                                </p>
                            </div>
                        )}
                        {activeStep === 1 && (
                            <div className="text-center mb-4">
                                <h3 className="fw-bold mb-2" style={{ fontFamily: "body", fontSize: '1.75rem' }}>Enter the contact details for a company admin and verify yourself</h3>
                            </div>
                        )}
                        {activeStep === 2 && (
                            <div className="text-center mb-4">
                                {/* OTP Verification / Success title and description handled in renderStepContent */}
                            </div>
                        )}

                        <CustomStepper
                            steps={['Organization', 'Contact', 'Verify']}
                            activeStep={activeStep}
                            mainColor={primaryColorForStepper} // Using the new variable here
                        />
                        <p className="text-center text-muted fw-bold mt-3 mb-4" style={{ fontFamily: "body", fontSize: '0.95rem' }}>
                            Step {activeStep + 1} of 3
                        </p>

                        <Formik
                            initialValues={initialValues}
                            validationSchema={registerValidationSchema}
                            onSubmit={handleSubmit}
                            validateOnChange={true}
                            validateOnBlur={true}
                        >
                            {({ values, handleChange, touched, errors, setTouched, isSubmitting }) => (
                                <FormikForm noValidate>
                                    {localError && <Alert variant="danger" className="mb-3 text-center">{localError}</Alert>}
                                    {authError && <Alert variant="danger" className="mb-3 text-center">{authError}</Alert>}

                                    {renderStepContent(values, handleChange, touched, errors)}

                                    <div className="d-flex justify-content-between mt-4">
                                        {activeStep > 0 && activeStep < 2 && (
                                            <Button
                                                variant="secondary"
                                                onClick={handleBack}
                                                className="px-4 py-2"
                                                style={{
                                                    backgroundColor: "#6c757d",
                                                    borderColor: "#6c757d",
                                                    color: "white",
                                                    boxShadow: "none",
                                                    fontFamily: "body",
                                                    fontWeight: "600"
                                                }}
                                            >
                                                <FontAwesomeIcon icon={faArrowLeft} className="me-2" /> Back
                                            </Button>
                                        )}

                                        {activeStep < 2 && (
                                            <Button
                                                type="button"
                                                onClick={() => handleNext(values, setTouched, errors)}
                                                className={`px-4 py-2 ${activeStep === 0 ? 'ms-auto' : ''}`}
                                                disabled={isSubmitting || isAnyLoading}
                                                style={{
                                                    backgroundColor: customStyles.primaryColor,
                                                    borderColor: customStyles.primaryColor,
                                                    color: "white",
                                                    boxShadow: "none",
                                                    fontFamily: "body",
                                                    fontWeight: "600",
                                                    opacity: (isSubmitting || isAnyLoading) ? 0.6 : 1
                                                }}
                                            >
                                                {activeStep === 0 && 'Next'}
                                                {activeStep === 1 && (isSubmitting ? <Spinner animation="border" size="sm" /> : 'Next')}
                                                <FontAwesomeIcon icon={faArrowRight} className="ms-2" />
                                            </Button>
                                        )}

                                        {activeStep === 2 && !isRegistrationSuccess && (
                                            <Button
                                                type="button"
                                                onClick={() => handleVerifyOtp(otp)}
                                                className="w-100 py-2 fw-semibold"
                                                disabled={otp.length !== OTP_LENGTH || isAnyLoading}
                                                style={{
                                                    backgroundColor: customStyles.primaryColor,
                                                    borderColor: customStyles.primaryColor,
                                                    color: "white",
                                                    boxShadow: "none",
                                                    fontFamily: "body",
                                                    opacity: (otp.length !== OTP_LENGTH || isAnyLoading) ? 0.6 : 1
                                                }}
                                            >
                                                {isLoading ? <Spinner animation="border" size="sm" className="me-2" /> : <FontAwesomeIcon icon={faCheckCircle} className="me-2" />} Verify & Sign Up
                                            </Button>
                                        )}

                                        {activeStep === 2 && isRegistrationSuccess && (
                                            <Button
                                                onClick={() => navigate('/login')}
                                                className="w-100 py-2 fw-semibold"
                                                style={{
                                                    backgroundColor: customStyles.primaryColor,
                                                    borderColor: customStyles.primaryColor,
                                                    color: "white",
                                                    boxShadow: "none",
                                                    fontFamily: "body",
                                                }}
                                            >
                                                Go to Login
                                            </Button>
                                        )}

                                    </div>

                                    {activeStep === 0 && (
                                        <div className="text-center mt-3">
                                            <p className="text-muted small" style={{ fontFamily: "body" }}>
                                                Already have an account?{" "}
                                                <Link to="/login" style={{ color: customStyles.primaryColor, textDecoration: "none", fontWeight: "600", fontFamily: "body" }}>
                                                    Sign in
                                                </Link>
                                            </p>
                                        </div>
                                    )}
                                    {activeStep === 1 && (
                                        <div className="text-center mt-3">
                                            <p className="text-muted small" style={{ fontFamily: "body" }}>
                                                Already have an account?{" "}
                                                <Link to="/login" style={{ color: customStyles.primaryColor, textDecoration: "none", fontWeight: "600", fontFamily: "body" }}>
                                                    Sign in
                                                </Link>
                                            </p>
                                        </div>
                                    )}
                                </FormikForm>
                            )}
                        </Formik>
                    </Card.Body>
                </Card>
            </Container>

            <footer style={{
                position: "fixed",
                bottom: 0,
                width: "100%",
                backgroundColor: customStyles.backgroundColor,
                padding: "15px 0",
                fontSize: "0.85rem",
                color: "#74787C",
                fontFamily: "body",
                zIndex: 100
            }}>
                <Container className="d-flex justify-content-between align-items-center">
                    <span>© 2025 Obeeoma. All rights reserved.</span>
                    <div>
                        <Link to="/privacy-policy" style={{ color: "#74787C", textDecoration: "none", marginRight: "20px" }}>
                            Privacy Policy
                        </Link>
                        <Link to="/terms-of-service" style={{ color: "#74787C", textDecoration: "none", marginRight: "20px" }}>
                            Terms of Service
                        </Link>
                        <Link to="/contact-us" style={{ color: "#74787C", textDecoration: "none" }}>
                            Contact Us
                        </Link>
                    </div>
                </Container>
            </footer>
        </div>
    );
};

export default Register;
